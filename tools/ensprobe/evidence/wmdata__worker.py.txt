"""Single owned cpuQ worker for S-WMDATA. No SUMO until the precharged ledger row exists.

Reuses the frozen E3/CTRLARM runtime and the byte-identical S-SPEEDHEAD arm wrappers
(tools/ctrlarm/zero.py, tools/speedhead/observer.py, tools/speedhead/baselines.py).
Arms: native (deploy --controller native), ctrl0/mappo (CTRLARM zero.py E3 path),
pi/circles (frozen deploy baselines). One thread, nice 19, CUDA hidden.
"""
from pathlib import Path
import contextlib
import fcntl
import hashlib
import importlib.metadata
import json
import os
import sys
import time
import traceback

OWN = Path('/home/ubuntu/wmdata20261031')
R = Path('/home/ubuntu/e3final')
SOURCE = 'b3c031adb6bc2b5c2bca160bea308f83ba5567b97c1cd0e853eddcd8367a8193'
MAPPO_EXPORT = '/home/ubuntu/mappospeed/production2/policy_latest.pt'
MAPPO_SHA = 'd9042cc5a05ee1adc54a720b7482329a59850bddb00444ce29cc70e49cffa9ca'
ZERO_SHA = '24095a1092eafb4c894804aa8142478a802eaea046142ec777e5f24f49f73596'
OBSERVER_SHA = 'b22af4623b5fe7fbb9a8920b8e8417c3591efae66a52bae7b72f4d899b4a7874'
BASELINES_SHA = 'c63b14bfaa49b993cade4608b0e69f714de6a286ea9cb1049816118541f4f3eb'
CTRL0_POLICY_SHA = '64f70af46adc582b426e789f082c19a8dca0525952d5b84e8fb18618b5f9fb5c'
ARMS = ('native', 'ctrl0', 'mappo', 'pi', 'circles')
sys.path[:0] = [str(OWN), str(R / 'runtime/src'), str(R / 'runtime'), str(OWN / 'tools/speedhead')]


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def emit(kind, **data):
    print('@@WMDATA ' + json.dumps(dict(kind=kind, **data), allow_nan=False), flush=True)


def write(p, d):
    with p.open('x') as f:
        json.dump(d, f, indent=1, allow_nan=False)
        f.write('\n')
        f.flush()
        os.fsync(f.fileno())


def verify():
    m = R / 'runtime/SOURCE_MANIFEST.json'
    assert sha(m) == SOURCE
    entries = json.loads(m.read_text())
    assert len(entries) == 543
    assert all(sha(R / 'runtime' / n) == h for n, h in entries.items())
    assert sha(OWN / 'tools/ctrlarm/zero.py') == ZERO_SHA
    assert sha(OWN / 'tools/speedhead/observer.py') == OBSERVER_SHA
    assert sha(OWN / 'tools/speedhead/baselines.py') == BASELINES_SHA
    assert sha(R / 'inputs/policy_latest.pt') == CTRL0_POLICY_SHA
    assert sha(MAPPO_EXPORT) == MAPPO_SHA


def charged_ids():
    rows = [json.loads(s) for s in (OWN / 'accounting.jsonl').read_text().splitlines()]
    assert len({r['id'] for r in rows}) == len(rows)
    assert sum(r['sim_s'] for r in rows) <= 180000
    return {r['id'] for r in rows}


def main():
    arm, seeds_file = sys.argv[1], Path(sys.argv[2])
    assert arm in ARMS
    assert os.getpriority(os.PRIO_PROCESS, 0) == 19
    assert os.environ['CUDA_VISIBLE_DEVICES'] == ''
    os.environ.setdefault('C4_POLICY_STOCHASTIC', '0')
    os.environ.setdefault('C4_POLICY_ACCEL_NOISE', '0')
    lock = (OWN / f'WORKER_{arm}_{seeds_file.stem}.lock').open('a')
    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
    import torch
    torch.set_num_threads(1)
    verify()
    assert importlib.metadata.version('eclipse-sumo') == '1.27.1'
    import shutil
    assert shutil.which('sumo'), 'SUMO binary not on PATH'
    from traffic_shepherd.config import load_config
    from traffic_shepherd.damping import CirclesController
    from traffic_shepherd.seeds import require_development_case
    from traffic_shepherd.c4.seeds import check_case
    from tools.ctrlarm.zero import run_e3
    from observer import Measurement, audit_lanes
    from wms.control.policy import load_policy_fn
    from baselines import run_baseline
    from traffic_shepherd.c4.env import CavSumoEnv
    from types import SimpleNamespace
    CirclesController(load_config())
    if arm in ('pi', 'circles'):
        fake = SimpleNamespace(cav_ids=frozenset({'preflight_cav'}), config=load_config(),
                               warmup_s=300, _circles=CirclesController(load_config()))
        CavSumoEnv._setup_controllers(fake, 'pi' if arm == 'pi' else 'damping')
        assert fake.fleet is not None
    if arm in ('native', 'ctrl0'):
        executor = load_policy_fn(R / 'inputs/policy_latest.pt', device='cpu')
        assert executor.residual_on_native
    design = json.loads((OWN / 'design.json').read_text())
    by_seed = {int(w['seed']): w for w in design['worlds']}
    seeds = [int(s) for s in seeds_file.read_text().split()]
    charged = charged_ids()
    emit('preflight', pid=os.getpid(), load=os.getloadavg(), source=SOURCE, arm=arm,
         seeds=seeds)
    for seed in seeds:
        w = by_seed[seed]
        require_development_case(seed, w['demand'], 'exposed')
        check_case(seed, w['demand'])
        job_id = f"s{seed}_{arm}"
        out = OWN / 'jobs' / job_id
        assert job_id in charged, 'attempt not precharged; refusing'
        assert not out.exists(), 'existing attempt; never retry'
        assert not (OWN / 'STOP').exists()
        while os.getloadavg()[0] > 28:
            emit('waiting_load', load=os.getloadavg())
            time.sleep(60)
            assert not (OWN / 'STOP').exists()
        verify()
        emit('start', job=job_id)
        out.mkdir(parents=True)
        started = dict(job=dict(id=job_id, seed=seed, arm=arm, penetration=w['penetration'],
                                demand=w['demand'], warmup=300, horizon=600, charge=900),
                       pid=os.getpid(), unix=time.time())
        write(out / 'START.json', started)
        m = Measurement()
        m.path = out
        begin = time.monotonic()
        try:
            with (out / 'worker.log').open('x') as log, \
                    contextlib.redirect_stdout(log), contextlib.redirect_stderr(log):
                with audit_lanes() as lanes:
                    runner = run_baseline if arm in ('pi', 'circles') else run_e3
                    result = runner(dict(started['job'], arm=arm), _inputs(), out, measurement=m)
            actual = float(m.progress)
            result.update(job=started['job'], actual_sim_s=actual,
                          wall_s=time.monotonic() - begin, lane_audit=lanes, ok=True,
                          source=SOURCE)
            assert actual == 900, actual
            assert all(v == 0 for v in lanes.values()), lanes
            assert set(result['metrics']['training_objective']) == {'120', '300', '600'}
            assert not any(result['metrics'][k] for k in
                           ('collisions', 'teleports', 'dropped_vehicle_steps'))
            if arm != 'native':
                ref = json.loads((OWN / 'jobs' / f"s{seed}_native" / 'result.json').read_text())
                assert result['metrics']['initial_state_sha256'] == \
                    ref['metrics']['initial_state_sha256']
            verify()
            result['output_hashes'] = {n: sha(out / n) for n in
                                       ('start.pt', 'start_history.npz', 'traj_base.npz',
                                        'deploy/episode/episode.npz')}
            write(out / 'result.json', result)
            emit('complete', job=job_id, actual_sim_s=actual,
                 wall_s=result['wall_s'])
        except BaseException:
            failure = dict(job=started['job'], actual_sim_s=float(getattr(m, 'progress', 0)),
                           charged_sim_s=900, traceback=traceback.format_exc())
            write(out / 'failure.json', failure)
            (OWN / 'STOP').write_text(job_id + '\n')
            emit('failed', **failure)
            raise
    emit('finished', arm=arm, jobs=len(seeds))


def _inputs():
    inputs = {k: str(R / 'inputs' / Path(v).name) for k, v in
              json.loads((R / 'inputs/INPUTS.json').read_text()).items()}
    inputs['mappo'] = MAPPO_EXPORT
    return inputs


if __name__ == '__main__':
    main()
